#################################################
# Core grammar / pipeline rules
#################################################

<query>            ::= <stmt> | <query> <stmt>

<stmt>             ::= <selectorDecl> | <directive>

<directive>        ::= '!' <directiveType>
                     | '!' <directiveType> '(' <directiveParams> ')'

<directiveType>    ::= 'import'

<selectorDecl>     ::= <selectorName> '=' <selectorPipeline>
                     | <selectorPipeline>

<selectorPipeline> ::= <selectorRefTuple>
                     | <selectorRefTuple> '|' <pipedDefs>

<pipedDefs>        ::= <selectorDef>
                     | <selectorDef> '|' <pipedDefs>

<selectorDef>      ::= <selectorType> '(' <params> ')'
                     | <selectorType> '(' ')'

<selectorRef>      ::= '%' <selectorName> | '%%'

<selectorRefTuple> ::= <selectorRef>
                     | '(' <selectorRefs> ')'

<selectorRefs>     ::= <selectorRef>
                     | <selectorRef> ',' <selectorRefs>

<selectorType>     ::= <identifier>

<params>           ::= <param>
                     | <params> ',' <param>

<param>            ::= <string> | <int> | <float> | <bool>

<selectorName>     ::= <identifier>

#################################################
# Lexical / helper rules
#################################################

<identifier>       ::= <letter> | <letter> <idRest>

<idRest>           ::= <idChar> | <idChar> <idRest>

<idChar>           ::= '_' | <letter> | <digit>

<letter>           ::= [a-z] | [A-Z]

<char>             ::= any character

<string>           ::= '"' <char>+ '"'

<bool>             ::= 'true' | 'false'

<digit>            ::= [0-9]

<int>              ::= '-' <digit>+ | <digit>+

<float>            ::= <neg_flo_
